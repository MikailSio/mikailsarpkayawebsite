C LANGUAGE

The foundation of systems programming. C gives you direct control over memory and hardware. Understanding C makes you a better programmer in any language — because it shows you what's really happening underneath.

01
Hello World & Compilation
main(), gcc, clang, compilation pipeline, header files
Available 5 min read
›

Every C program starts with main(). Before your code runs, the compiler takes your .c source file through four distinct stages: preprocessing, compilation, assembling, and linking.

C hello.c
#include <stdio.h>   /* standard input/output header */

int main(void) {
    printf("Hello, World!
");
    return 0;            /* 0 = success */
}
BASH Compilation & Run
# Basic compile
gcc hello.c -o hello
./hello

# Recommended: enable all warnings
gcc -Wall -Wextra -std=c11 hello.c -o hello

# With debug symbols (for gdb)
gcc -g -Wall hello.c -o hello

The four compilation stages:

  • Preprocessing — expands #include, #define macros (produces .i file)
  • Compilation — translates C to assembly code (produces .s file)
  • Assembling — converts assembly to machine code object file (.o)
  • Linking — combines object files + libraries into final executable
Tip: Always compile with -Wall -Wextra. Warnings are bugs waiting to happen. Fix every warning before moving on.
C Command-line arguments
#include <stdio.h>

int main(int argc, char *argv[]) {
    printf("Arguments: %d
", argc);
    for (int i = 0; i < argc; i++) {
        printf("  argv[%d] = %s
", i, argv[i]);
    }
    return 0;
}
/* Run: ./hello Mike 42  →  argv[0]=./hello, argv[1]=Mike ... */
02
Variables & Data Types
int, float, double, char, sizeof, limits, type casting
Available 8 min read
›

C is a statically typed language — every variable must be declared with a type before use. The type determines how many bytes are allocated and how the bits are interpreted.

C Primitive types
#include <stdio.h>

int main(void) {
    /* Integer types */
    char           c  = 'A';        /* 1 byte,  -128 to 127 */
    short          s  = 32000;      /* 2 bytes */
    int            i  = 2147483647; /* 4 bytes */
    long           l  = 9999999L;   /* 8 bytes on 64-bit */
    long long      ll = 9223372036854775807LL;

    /* Floating point */
    float  f  = 3.14f;         /* 4 bytes, ~7 sig digits */
    double d  = 3.14159265358; /* 8 bytes, ~15 sig digits */

    /* Unsigned (no negative values, double positive range) */
    unsigned int u = 4294967295U;

    /* sizeof — bytes at compile time */
    printf("int=%zu  double=%zu  char=%zu
",
           sizeof(int), sizeof(double), sizeof(char));

    return 0;
}
C Type casting & integer division trap
int a = 7, b = 2;

/* INTEGER division — truncates! */
printf("%d
", a / b);             /* 3, not 3.5 */

/* Cast one operand to get float division */
printf("%.2f
", (double)a / b);   /* 3.50 */

/* Implicit narrowing — dangerous */
double pi = 3.99;
int    n  = (int)pi;  /* n = 3, decimal truncated */
  • Prefer double over float — float has only ~7 decimal digits of precision
  • Use unsigned when a value can never be negative (array indices, sizes)
  • sizeof returns size_t — use %zu format specifier, not %d
C Fixed-width types (prefer these)
#include <stdint.h>  /* exact-width types */

int8_t   a = 127;          /* exactly 8 bits  */
int16_t  b = 32767;        /* exactly 16 bits */
int32_t  c = 2147483647;   /* exactly 32 bits */
int64_t  d = 9223372036LL; /* exactly 64 bits */
uint32_t e = 4294967295U;  /* unsigned 32 bits */
Rule: Use int32_t, uint64_t etc. from <stdint.h> in production code. The size of int is platform-dependent; fixed-width types are guaranteed.
03
Pointers & Memory
*, &, malloc, calloc, realloc, free, NULL, memory leaks
Available 12 min read
›

A pointer is a variable that stores a memory address. This is the most powerful — and most dangerous — feature of C. Master pointers and you master C.

C Pointer basics
#include <stdio.h>

int main(void) {
    int  x   = 42;
    int *ptr = &x;       /* & = address-of operator */

    printf("Value:   %d
", x);    /* 42          */
    printf("Address: %p
", ptr);  /* 0x7ffd...   */
    printf("Deref:   %d
", *ptr); /* 42 (deref)  */

    *ptr = 100;  /* modify x through pointer */
    printf("x is now: %d
", x);  /* 100 */

    return 0;
}
C Passing by pointer (simulate pass-by-reference)
/* Swap two integers — classic pointer example */
void swap(int *a, int *b) {
    int tmp = *a;
    *a = *b;
    *b = tmp;
}

int main(void) {
    int x = 5, y = 10;
    swap(&x, &y);
    printf("%d %d
", x, y);  /* 10 5 */
    return 0;
}
C Dynamic memory allocation
#include <stdio.h>
#include <stdlib.h>   /* malloc, calloc, realloc, free */

int main(void) {
    int n = 5;

    /* malloc — allocate, uninitialized */
    int *arr = (int *)malloc(n * sizeof(int));
    if (arr == NULL) {
        perror("malloc failed");
        return 1;
    }

    for (int i = 0; i < n; i++) arr[i] = i * 10;

    /* calloc — allocate + zero-initialize */
    int *zeros = (int *)calloc(n, sizeof(int));

    /* realloc — resize existing allocation */
    arr = (int *)realloc(arr, 10 * sizeof(int));

    free(arr);    /* always free! */
    free(zeros);
    arr = NULL;   /* good habit: null the pointer after free */

    return 0;
}
Common pointer bugs: forgetting to free (memory leak), using after free (undefined behavior), double free (crash), null pointer dereference. Use valgrind ./program to detect leaks.
C Pointer arithmetic
int arr[] = {10, 20, 30, 40, 50};
int *p = arr;         /* points to arr[0] */

printf("%d
", *p);      /* 10 */
printf("%d
", *(p+2));  /* 30 — moves 2*sizeof(int) bytes */
printf("%d
", p[3]);    /* 40 — array indexing = pointer arithmetic */

p++;                       /* now points to arr[1] */
04
Functions & Recursion
prototypes, pass-by-value, call stack, recursion, function pointers
Available 10 min read
›

C is pass-by-value — functions always receive copies of arguments. To modify the caller's variables, pass a pointer. Functions must be declared before they are used (forward declaration / prototype).

C Function prototype + definition
#include <stdio.h>

/* Prototype — tells compiler the signature */
double power(double base, int exp);
void   print_array(int *arr, int n);

int main(void) {
    printf("2^10 = %.0f
", power(2.0, 10));  /* 1024 */
    int data[] = {1, 2, 3, 4, 5};
    print_array(data, 5);
    return 0;
}

double power(double base, int exp) {
    double result = 1.0;
    for (int i = 0; i < exp; i++) result *= base;
    return result;
}

void print_array(int *arr, int n) {
    for (int i = 0; i < n; i++)
        printf("%d ", arr[i]);
    printf("
");
}
C Recursion — factorial & fibonacci
long long factorial(int n) {
    if (n <= 1) return 1;        /* base case */
    return n * factorial(n - 1); /* recursive case */
}

/* Tail-recursive style (iterative equivalent) */
long long fact_iter(int n) {
    long long acc = 1;
    while (n > 1) acc *= n--;
    return acc;
}

printf("10! = %lld
", factorial(10));  /* 3628800 */
C Function pointers — callbacks
#include <stdlib.h>

/* Function pointer type: takes two const void*, returns int */
int cmp_int(const void *a, const void *b) {
    return *(const int*)a - *(const int*)b;
}

int data[] = {5, 1, 4, 2, 8};
qsort(data, 5, sizeof(int), cmp_int);
/* data is now: 1 2 4 5 8 */
Stack depth: Deeply recursive calls can cause a stack overflow. For large inputs, convert recursion to iteration with an explicit stack (array).
05
Arrays & Strings
stack arrays, char arrays, null terminator, string.h, multi-dim arrays
Available 10 min read
›

In C, an array is a contiguous block of memory. The array name decays to a pointer to its first element. Strings are just char arrays that end with a null terminator ''.

C Arrays
#include <stdio.h>

int main(void) {
    /* Fixed-size array on stack */
    int scores[5] = {85, 92, 78, 95, 88};

    /* Length at compile time only */
    int len = sizeof(scores) / sizeof(scores[0]);  /* 5 */

    int sum = 0;
    for (int i = 0; i < len; i++) sum += scores[i];
    printf("Average: %.1f
", (double)sum / len);

    /* 2D array — matrix */
    int mat[3][3] = {
        {1, 2, 3},
        {4, 5, 6},
        {7, 8, 9}
    };
    printf("mat[1][2] = %d
", mat[1][2]);  /* 6 */

    return 0;
}
C Strings — char arrays
#include <string.h>

char name[] = "Mikail";       /* 7 bytes: M i k a i l  */
printf("Length: %zu
", strlen(name));  /* 6 (not 7, no ) */

/* String copy — ALWAYS check buffer size */
char buf[50];
strncpy(buf, name, sizeof(buf) - 1);
buf[sizeof(buf) - 1] = '';       /* ensure null termination */

/* Concatenation */
strncat(buf, " Sarpkaya", sizeof(buf) - strlen(buf) - 1);

/* Comparison */
if (strcmp(name, "Mikail") == 0)
    printf("Names match!
");

/* Search */
char *pos = strstr(buf, "Sarp");
if (pos) printf("Found at offset: %td
", pos - buf);
Never use: strcpy, strcat, gets — they don't check buffer bounds and cause buffer overflows. Always use the n-bounded variants: strncpy, strncat, fgets.
06
File I/O in C
fopen, fclose, fprintf, fscanf, fgets, fread, fwrite, binary files
Available 8 min read
›

File I/O in C uses FILE pointers. Always check that fopen succeeds (returns non-NULL), and always call fclose when done.

C Write & read text file
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    /* --- WRITE --- */
    FILE *fw = fopen("data.txt", "w");
    if (!fw) { perror("fopen"); return 1; }

    fprintf(fw, "Name: %s
", "Mikail");
    fprintf(fw, "Score: %d
", 95);
    fclose(fw);

    /* --- READ line by line --- */
    FILE *fr = fopen("data.txt", "r");
    if (!fr) { perror("fopen"); return 1; }

    char line[256];
    while (fgets(line, sizeof(line), fr) != NULL)
        printf("Read: %s", line);

    fclose(fr);
    return 0;
}
C Binary file — write & read struct
typedef struct {
    char  name[32];
    float score;
    int   year;
} Record;

/* Write binary */
Record r = {"Mikail", 95.5f, 2025};
FILE *fb = fopen("rec.bin", "wb");
fwrite(&r, sizeof(Record), 1, fb);
fclose(fb);

/* Read binary */
Record r2;
fb = fopen("rec.bin", "rb");
fread(&r2, sizeof(Record), 1, fb);
fclose(fb);
printf("%s %.1f %d
", r2.name, r2.score, r2.year);
  • "r" — read text, "w" — write text (truncates), "a" — append text
  • "rb" — read binary, "wb" — write binary
  • "r+" — read & write existing file
  • Use fseek + ftell to navigate within a file
Always: check fopen return value, close files with fclose, and use fgets not scanf for reading strings from files.
