Sets — Unique Elements

Python
stopwords = {"the", "is", "a", "an", "on"}
tokens    = {"the", "cat", "sat", "on"}

# Set operations
print(tokens - stopwords)     # {'cat', 'sat'}   — difference
print(tokens & stopwords)     # {'the', 'on'}    — intersection
print(tokens | stopwords)     # union of both
print(tokens.issubset(stopwords))  # False

# Modify
stopwords.add("in")           # add item
stopwords.remove("an")        # remove (KeyError if missing)
stopwords.discard("xyz")      # remove (no error if missing)

# Deduplicate a list
words = ["the", "cat", "the", "sat"]
unique = list(set(words))    # ['the', 'cat', 'sat']
Output: 
{'cat', 'sat'}
{'the', 'on'}
When to use which: list — ordered, allows duplicates. tuple — immutable, hashable (dict keys). dict — fast key lookup O(1). set — unique items, fast membership test O(1).

def compute_f1(precision: float, recall: float) -> float:
    """Compute F1 score from precision and recall."""
    if precision + recall == 0:
        return 0.0
    return 2 * (precision * recall) / (precision + recall)

Scope (LEGB Rule) & Closures

Python
# LEGB: Local → Enclosing → Global → Built-in
x = "global"

def outer():
    x = "enclosing"
    def inner():
        x = "local"
        print(x)    # "local"
    inner()
    print(x)        # "enclosing"

outer()
print(x)            # "global"

# Closure — function that remembers its enclosing scope
def make_multiplier(factor):
    def multiply(x):
        return x * factor
    return multiply

double = make_multiplier(2)
triple = make_multiplier(3)
print(double(5))   # 10
print(triple(5))   # 15
Output: 
local
enclosing
global
10
15

Decorators

Python
import time
from functools import wraps

# Decorator — wraps a function with extra behavior
def timer(func):
    @wraps(func)   # preserves original function metadata
    def wrapper(*args, **kwargs):
        start = time.time()
        result = func(*args, **kwargs)
        elapsed = time.time() - start
        print(f"{func.__name__} took {elapsed:.4f}s")
        return result
    return wrapper

@timer
def preprocess(texts):
    return [t.lower().strip() for t in texts]

result = preprocess(["  Hello  ", "WORLD"])
print(result)
Output: 
preprocess took 0.0000s
['hello', 'world']
Never use a mutable default: def func(items=[]) shares the same list across calls. Use def func(items=None) and set items = items or [] inside.

Object-Oriented Programming is how PyTorch, HuggingFace, and Scikit-learn are built. Every model is a class, every dataset is a class. Understanding OOP means understanding the tools you use daily in ML.

Classes & Dunder Methods

Python
class TextDataset:
    """A simple text dataset for NLP tasks."""

    # Class attribute (shared by all instances)
    version = "1.0"

    def __init__(self, texts: list[str], labels: list[int]):
        # Instance attributes
        self.texts  = texts
        self.labels = labels

    def __len__(self):
        return len(self.texts)

    def __repr__(self):
        return f"TextDataset({len(self)} samples)"

    def __getitem__(self, idx):
        return self.texts[idx], self.labels[idx]

    def __eq__(self, other):
        return self.texts == other.texts and self.labels == other.labels

    @property
    def num_classes(self):
        return len(set(self.labels))

    @classmethod
    def from_file(cls, path):
        # Alternative constructor
        texts, labels = [], []
        with open(path) as f:
            for line in f:
                t, l = line.strip().split("\t")
                texts.append(t)
                labels.append(int(l))
        return cls(texts, labels)

    @staticmethod
    def tokenize(text):
        return text.lower().split()

# Usage
ds = TextDataset(["great movie", "bad film", "ok"], [1, 0, 1])
print(ds)                # TextDataset(3 samples)
print(len(ds))            # 3
print(ds[0])              # ('great movie', 1)
print(ds.num_classes)     # 2
print(TextDataset.tokenize("Hello World"))
Output: 
TextDataset(3 samples)
3
('great movie', 1)
2
['hello', 'world']

Robust code handles errors gracefully. In ML projects, you'll encounter FileNotFoundError when loading datasets, ValueError when shapes mismatch, and KeyError when config keys are missing. Proper exception handling makes debugging faster.

try / except / else / finally

Python
# Basic pattern
try:
    data = open("corpus.txt").read()
except FileNotFoundError:
    print("File not found — using empty corpus")
    data = ""
except (PermissionError, IOError) as e:
    print(f"IO error: {e}")
    data = ""
except Exception as e:
    print(f"Unexpected error: {e}")
    data = ""
else:
    print(f"Loaded {len(data)} characters")  # runs only if NO exception
finally:
    print("Cleanup complete")              # ALWAYS runs
Output: 
File not found — using empty corpus
Cleanup complete

# Raise exceptions explicitly
def load_model(name: str):
    valid = ["bert", "roberta", "lstm"]

# Custom exception class
class DataPipelineError(Exception):
    """Raised when data pipeline fails."""
    pass

class ShapeMismatchError(DataPipelineError):
    """Raised when tensor shapes don't match."""
    def __init__(self, expected, got):
        super().__init__(f"Expected shape {expected}, got {got}")

# Usage
def validate_batch(inputs, labels):
    if len(inputs) != len(labels):
        raise ShapeMismatchError(len(inputs), len(labels))

try:
    validate_batch([1, 2, 3], [0, 1])
except ShapeMismatchError as e:
    print(e)
Output: 
Expected shape 3, got 2

from contextlib import contextmanager

@contextmanager
def timer_context(label):
    import time
    start = time.time()
    yield

with timer_context("Preprocessing"):

Generators produce items one at a time using yield instead of building entire lists in memory. This is essential when processing large NLP corpora — you can iterate through millions of text samples without loading them all into RAM.

Iterators — iter() & next()

Python
# Any iterable can be turned into an iterator
nums = [10, 20, 30]
it = iter(nums)

print(next(it))   # 10
print(next(it))   # 20
print(next(it))   # 30
# next(it)        # StopIteration error

# Custom iterator class
class Countdown:
    def __init__(self, start):
        self.n = start
    def __iter__(self):
        return self
    def __next__(self):
        if self.n <= 0:
            raise StopIteration
        self.n -= 1
        return self.n + 1

for x in Countdown(3):
    print(x, end=" ")

Generator Functions & Expressions

Python
# Generator function — yields one item at a time
def read_lines(filepath):
    """Memory-efficient line reader."""
    with open(filepath, encoding="utf-8") as f:
        for line in f:
            yield line.strip()

# Usage — one line at a time, no full list in memory
for line in read_lines("corpus.txt"):
    tokens = line.split()

# Generator expression — lazy version of list comprehension
squares_list = [x**2 for x in range(1_000_000)]  # ~8MB RAM
squares_gen  = (x**2 for x in range(1_000_000))  # ~0 RAM

# sum/max/min work directly with generators
total = sum(x**2 for x in range(100))  # no list created

Batch Generator & Processing Pipelines

# ★ Batch Generator — critical pattern for ML training
def batch_generator(data, batch_size=32):
    """Yield successive batches from data."""
    for i in range(0, len(data), batch_size):
        yield data[i : i + batch_size]

texts = [f"sample {i}" for i in range(100)]
for batch in batch_generator(texts, batch_size=32):
    print(f"Batch of {len(batch)} items")

# Chain generators — build processing pipelines
def clean_texts(lines):
    for line in lines:
        cleaned = line.strip().lower()
        if cleaned:
            yield cleaned

def tokenize(texts):
    for text in texts:
        yield text.split()

Regular expressions are the surgeon's scalpel of text processing. In NLP pipelines, they handle URL removal, number normalization, special character cleaning, and language-specific preprocessing — especially important for Turkish text with its unique characters.

Regex Basics — search, match, findall

import re

text = "Contact us at info@example.com or visit https://example.com"

# search — first match anywhere in string
m = re.search(r'\d+', "epoch 5, loss 0.03")
print(m.group())     # "5"

# match — match at START of string only
m = re.match(r'epoch', "epoch 5")
print(m.group())     # "epoch"

# findall — all matches as list
emails = re.findall(r'[\w.+-]+@[\w-]+\.[\w.]+', text)
print(emails)        # ['info@example.com']

urls = re.findall(r'https?://\S+', text)
print(urls)          # ['https://example.com']

# Common patterns
digits   = re.findall(r'\d+', "epoch 5, loss 0.03")     # ['5', '0', '03']
words    = re.findall(r'\b\w+\b', "Hello, World!")      # ['Hello', 'World']
hashtags = re.findall(r'#\w+', "#NLP #AI #Python")     # ['#NLP', '#AI', '#Python']

Pattern Reference

Python
# ── Character classes ──
# \d  → digit [0-9]
# \w  → word char [a-zA-Z0-9_]
# \s  → whitespace [ \t\n\r]
# \D, \W, \S → negations
# .   → any character (except newline)

# ── Quantifiers ──
# *   → 0 or more
# +   → 1 or more
# ?   → 0 or 1 (optional)
# {n} → exactly n
# {n,m} → n to m

# ── Anchors ──
# ^   → start of string
# $   → end of string
# \b  → word boundary

# ── Groups ──
m = re.search(r'(\w+)@(\w+)\.(\w+)', "user@gmail.com")
print(m.group(0))   # user@gmail.com  (full match)
print(m.group(1))   # user
print(m.group(2))   # gmail

# Named groups
m = re.search(r'(?P<user>\w+)@(?P<domain>\w+)', "user@gmail.com")

NLP Text Cleaning Pipeline

Python
import re
import unicodedata

def clean_text(text: str, lang: str = "tr") -> str:
    """Clean text for NLP pipeline.
    Handles Turkish characters, removes noise,
    normalizes whitespace."""

    # Normalize unicode (important for Turkish ı/İ, ö/Ö)
    text = unicodedata.normalize("NFC", text)

    # Turkish-aware lowercase
    if lang == "tr":
        text = text.replace("I", "ı").replace("İ", "i")
    text = text.lower()

    # Remove URLs
    text = re.sub(r'https?://\S+', '', text)

    # Remove emails
    text = re.sub(r'[\w.+-]+@[\w-]+\.[\w.]+', '', text)

    # Remove HTML tags
    text = re.sub(r'<[^>]+>', '', text)

    # Keep only letters, digits, spaces, Turkish chars
    text = re.sub(r'[^\w\sçğıöşüÇĞİÖŞÜ]', ' ', text)

    # Normalize whitespace
    text = re.sub(r'\s+', ' ', text).strip()

    return text

# Usage
raw = "  <p>Merhaba Dünya! 🌍 Visit https://example.com  </p>  "
print(clean_text(raw))

# Batch processing
corpus = ["<b>Text 1</b>", "  Text 2  ", "Text @email.com 3"]
cleaned = [clean_text(t) for t in corpus]
print(cleaned)
Output: 
merhaba dünya visit
['text 1', 'text 2', 'text 3']
Turkish I/ı Problem: Python's .lower() converts I → i, but in Turkish I → ı and İ → i. Always handle this explicitly when processing Turkish text — it directly affects tokenization and model accuracy.
